from flask import Flask, jsonify
from flask_cors import CORS
import os, json, re

app = Flask(__name__)
CORS(app)

# AI patterns to detect
AI_PATTERNS = {
    "openai": [r"import openai", r"from openai", r"OpenAI\(\)"],
    "anthropic": [r"import anthropic", r"from anthropic"],
    "google": [r"import google.generativeai", r"genai\."],
    "langchain": [r"from langchain", r"import langchain"],
    "huggingface": [r"from transformers", r"pipeline\("],
    "cohere": [r"import cohere", r"cohere\.Client"],
}

MODEL_PATTERNS = {
    "gpt-4": r"gpt-4",
    "gpt-3.5": r"gpt-3\.5",
    "claude": r"claude",
    "gemini": r"gemini",
    "llama": r"llama",
}

def scan_directory(root_path):
    result = {"folders": [], "agents": []}
    for dirpath, dirnames, filenames in os.walk(root_path):
        folder_info = {
            "path": dirpath,
            "name": os.path.basename(dirpath),
            "subfolders": dirnames,
            "files": filenames,
            "agent_count": 0
        }
        for f in filenames:
            if f.endswith(('.py','.js','.ts','.java','.cs')):
                fpath = os.path.join(dirpath, f)
                try:
                    content = open(fpath).read()
                    for provider, patterns in AI_PATTERNS.items():
                        for p in patterns:
                            if re.search(p, content):
                                model = "unknown"
                                for m, mp in MODEL_PATTERNS.items():
                                    if re.search(mp, content):
                                        model = m
                                result["agents"].append({
                                    "script": f,
                                    "path": fpath,
                                    "provider": provider,
                                    "model": model,
                                    "folder": dirpath
                                })
                                folder_info["agent_count"] += 1
                                break
                        else: continue
                        break
                except: pass
        result["folders"].append(folder_info)
    return result

@app.route('/api/scan')
def scan():
    path = request.args.get('path', '.')
    return jsonify(scan_directory(path))

@RestController
@CrossOrigin(origins = "*")
@RequestMapping("/api")
public class AgentScannerController {

    @GetMapping("/scan")
    public ResponseEntity<Map> scanProject(
        @RequestParam(defaultValue=".")
        String path) {

        File root = new File(path);
        Map<String, Object> result = new HashMap<>();
        List<Map> agents = new ArrayList<>();
        List<Map> folders = new ArrayList<>();

        scanDir(root, agents, folders);

        result.put("agents", agents);
        result.put("folders", folders);
        return ResponseEntity.ok(result);
    }

    private void scanDir(File dir,
        List<Map> agents, List<Map> folders) {

        if (!dir.isDirectory()) return;
        File[] files = dir.listFiles();
        Map<String,Object> folderInfo = new HashMap<>();
        folderInfo.put("name", dir.getName());
        folderInfo.put("path", dir.getPath());
        int agentCount = 0;

        for (File f : files) {
            if (f.isDirectory()) {
                scanDir(f, agents, folders);
            } else {
                String content = readFile(f);
                if (content.contains("openai")
                    || content.contains("anthropic")
                    || content.contains("genai")) {
                    Map<String,String> agent = new HashMap<>();
                    agent.put("script", f.getName());
                    agent.put("path", f.getPath());
                    agents.add(agent);
                    agentCount++;
                }
            }
        }
        folderInfo.put("agentCount", agentCount);
        folders.add(folderInfo);
    }

    @GetMapping("/agents")
    public ResponseEntity<?> getAgents() {
        return ResponseEntity.ok(agentService.getAll());
    }
}

// AgentScannerController.cs
using Microsoft.AspNetCore.Mvc;
using System.Text.RegularExpressions;

[ApiController]
[Route("api")]
public class AgentScannerController
    : ControllerBase
{
    private static readonly Dictionary<string,string[]>
    Patterns = new() {
        ["openai"] = new[]{"openai","OpenAI"},
        ["anthropic"] = new[]{"anthropic","Claude"},
        ["google"] = new[]{"genai","Gemini"},
    };

    [HttpGet("scan")]
    public IActionResult Scan(
        [FromQuery] string path = ".")
    {
        var agents = new List<object>();
        var folders = new List<object>();

        ScanDirectory(path, agents, folders);

        return Ok(new {
            agents, folders
        });
    }

    private void ScanDirectory(
        string dir,
        List<object> agents,
        List<object> folders)
    {
        var agentCount = 0;
        foreach (var file in
            Directory.GetFiles(dir))
        {
            var content = System.IO.File
                .ReadAllText(file);
            foreach (var kvp in Patterns)
            {
                if (kvp.Value.Any(p =>
                    content.Contains(p)))
                {
                    agents.Add(new {
                        script = Path.GetFileName(file),
                        provider = kvp.Key,
                        path = file
                    });
                    agentCount++;
                    break;
                }
            }
        }
        folders.Add(new {
            name = new DirectoryInfo(dir).Name,
            path = dir,
            agentCount
        });
        foreach (var sub in
            Directory.GetDirectories(dir))
            ScanDirectory(sub, agents, folders);
    }

    [HttpGet("agents")]
    public IActionResult GetAgents()
        => Ok(AgentService.GetAll());
}
